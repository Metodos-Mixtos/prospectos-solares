% Lamina de transicion entre dos pasos. Siempre la misma composicion: a la
% izquierda el resultado que sale, a la derecha el paso que lo recibe, y debajo el
% parrafo que explica por que ese resultado basta.
%   #1 rotulo del resultado   #2 contenido del resultado

% --- 15. Los limites del modelo ---
% Media lamina de limites, dicha en voz alta y antes de que la pregunte nadie.
%
% El tercer parrafo repetia palabra por palabra el segundo de la lamina siguiente
% («filtro deliberadamente estrecho: prioriza conexion barata sobre cobertura del
% territorio»). Queda solo donde le corresponde, que es la lamina del radio.
\begin{frame}{Qué no hace el modelo}{Los límites, dichos antes de que los pregunten}
  \vspace{0.6em}
  \hspace{0.01\textwidth}\begin{minipage}{0.96\textwidth}
    \raggedright\small\textcolor{tinta2}{\textbf{No hay medida de precisión fuera de
    muestra.} El puntaje ordena el territorio por parecido con lo ya construido; no
    estima probabilidad de éxito, ni de conexión, ni de rentabilidad. No hay
    evaluación contra un conjunto reservado, y por tanto no se puede citar un acierto.

    \vspace{1.0em}
    \textbf{La ponderación entre los dos componentes está puesta a criterio}, 60 y 40,
    y no sale de ninguna medición. Es el único parámetro del procedimiento que se
    eligió a mano: los siete pesos del paso 4 y sus umbrales sí están medidos.}
  \end{minipage}
  \fuente{Los estudios de precisión y de prueba que existen son legado de un trabajo
  anterior y no validan este modelo.}
\end{frame}

% --- 16 ---
\begin{frame}{El radio de la subestación es un filtro previo, no un criterio}
  \afirmacion{El modelo no puntúa las 21.447 grillas del país. Puntúa solo las que caen
  dentro del radio de una subestación y aún no tienen planta.}
  {Ese radio no se fijó a mano: es el promedio de la distancia de cada planta solar
  existente a su subestación más cercana. Al ser la media, deja dentro 71 de las 112
  grillas con planta y fuera 41. Es un filtro deliberadamente estrecho, que prioriza
  conexión barata sobre cobertura del territorio.}

  \vspace{0.6em}
  \hspace{0.01\textwidth}\begin{minipage}{0.96\textwidth}
    \raggedright
    \cifraancla[0.32\linewidth]{16,2 km}{la media entre cada planta y su subestación}\hfill
    \cifraancla[0.30\linewidth]{7.239}{grillas dentro del radio}\hfill
    \cifraancla[0.32\linewidth]{71 de 112}{grillas con planta que quedan dentro}
  \end{minipage}
  \fuente{La Orinoquía y la Amazonía no aparecen por este radio, no por calificación baja.}
\end{frame}
